// Chama Viva / Abrigo 77. Four original maps, shared reconstruction and per-character rewards.
// Persistent $ variables belong to the world. er_city_done belongs to each character.
function	script	F_ChamaMap	{
	return "er_chama" + $er_city_stage;
}

-	script	ChamaVivaGuard	-1,{
OnInit:
	if ($er_city_stage < 0 || $er_city_stage > 3) $er_city_stage = 0;
	end;
OnPCLoadMapEvent:
	.@map$ = strcharinfo(3);
	if (.@map$ != "er_chama0" && .@map$ != "er_chama1" && .@map$ != "er_chama2" && .@map$ != "er_chama3") end;
	if (.@map$ != callfunc("F_ChamaMap")) {
		warp callfunc("F_ChamaMap"),48,28;
		end;
	}
	getmapxy .@current$, .@x, .@y;
	if (!checkcell(.@map$, .@x, .@y, cell_chkpass)) { warp callfunc("F_ChamaMap"),48,28; end; }
	if (!er_city_intro) {
		er_city_intro = 1;
		dispbottom "Chama Viva: procure Mara Voss na praca. A oficina, a estufa e a torre precisam de ajuda.";
	}
	end;
}

er_chama0,22,39,5	script	Oficina#cv0	4_M_ALCHE_A,{
	mes "[Ivo - Oficina de sucata]";
	mes "Os galhos cobriram nossos modulos de Helios. Precisamos de dez pecas de sucata (Jellopy) para recuperar a oficina.";
	if (er_city_done & 1) {mes "Sua contribuicao ja foi registrada. Obrigado, Desperto.";close;}
	mes "Recupere material nos tres depositos da vila ou cace Porings nos campos de Prontera.";
	if (select("Entregar 10 Jellopy:Depois") == 2) close;
	// Menus suspend the script; revalidate all conditions after the selection.
	if (er_city_done & 1) close;
	if (countitem("Jellopy") < 10 || !checkweight("Red_Potion",5)) {mes "Traga dez Jellopy e reserve espaco para cinco pocoes.";close;}
	er_city_done |= 1;
	delitem "Jellopy",10;
	getitem "Red_Potion",5;
	$er_city_scrap += 10;
	if ($er_city_stage == 0) {
		$er_city_stage = 1;
		announce "Chama Viva: a oficina foi recuperada. A floresta e a tecnologia podem coexistir!",bc_all;
		close2;
		mapwarp "er_chama0","er_chama1",48,28;
		end;
	}
	mes "Contribuicao registrada. Sua recompensa foi entregue.";
	close;
}
