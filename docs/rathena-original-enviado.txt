// 1. MARA VOSS - Prologo e escolha da classe inicial
//    Variavel eridan_arch: 1 Guerreiro, 2 Arqueiro, 3 Tank,
//    4 Mago de Luz, 5 Feiticeiro de Trevas
// ---------------------------------------------------------
prontera,155,185,5	script	Mara Voss#eridan	4_F_KAFRA1,{
	mes "[Mara Voss]";
	if (eridan_arch) {
		mes "Voce ja escolheu seu caminho, Desperto.";
		mes "Volte quando for hora da Prova de Ascensao.";
		close;
	}
	if (Class != Job_Novice) {
		mes "Seu Marcador de Eter nao reage mais a escolha inicial.";
		close;
	}
	mes "Voce acordou no Abrigo 77, Desperto.";
	mes "A Ruptura destruiu Helios e o Eter se partiu em Lumen e Umbra.";
	mes "Minha guilda, a Chama Viva, precisa de agua e energia.";
	mes "Ajude-nos e eu ensino voce a sobreviver.";
	next;
	if (JobLevel < 10) {
		mes "[Mara Voss]";
		mes "Antes, treine ate o nivel de classe 10.";
		mes "Seu Marcador de Eter ainda esta instavel.";
		close;
	}
	mes "[Mara Voss]";
	mes "Escolha seu caminho:";
	switch(select("Guerreiro:Arqueiro:Tank:Mago de Luz (Lumen):Feiticeiro de Trevas (Umbra)")) {
	case 1:
		set eridan_arch, 1;
		jobchange Job_Swordman;
		break;
	case 2:
		set eridan_arch, 2;
		jobchange Job_Archer;
		break;
	case 3:
		set eridan_arch, 3;
		jobchange Job_Swordman;
		break;
	case 4:
		set eridan_arch, 4;
		jobchange Job_Acolyte;
		break;
	case 5:
		set eridan_arch, 5;
		jobchange Job_Mage;
		break;
	}
	next;
	mes "[Mara Voss]";
	mes "Bem-vindo a Chama Viva.";
	mes "Procure o Mestre de Ascensao no nivel 30.";
	close;
}

// ---------------------------------------------------------
// 2. PROVA DE ASCENSAO - Promocao de classe (2 opcoes por classe)
//    Requer BaseLevel 30 e JobLevel 40
// ---------------------------------------------------------
prontera,158,185,5	script	Mestre de Ascensao#eridan	4_M_SAGE_C,{
	mes "[Mestre de Ascensao]";
	if (!eridan_arch) {
		mes "Fale primeiro com Mara Voss.";
		close;
	}
	if (BaseLevel < 30 || JobLevel < 40) {
		mes "A Prova exige nivel base 30 e nivel de classe 40.";
		close;
	}
	if (Class == Job_Knight || Class == Job_Assassin || Class == Job_Hunter ||
	    Class == Job_Rogue || Class == Job_Crusader || Class == Job_Blacksmith ||
	    Class == Job_Priest || Class == Job_Monk || Class == Job_Wizard || Class == Job_Sage) {
		mes "Voce ja concluiu sua Ascensao.";
		close;
	}
	mes "Escolha sua promocao:";
	switch(eridan_arch) {
	case 1:
		switch(select("Gladiador de Guerra (Cavaleiro):Lamina-Fantasma (Assassino)")) {
		case 1: jobchange Job_Knight; break;
		case 2: jobchange Job_Assassin; break;
		}
		break;
	case 2:
		switch(select("Atirador de Elite (Cacador):Cacador de Rastros (Gatuno)")) {
		case 1: jobchange Job_Hunter; break;
		case 2: jobchange Job_Rogue; break;
		}
		break;
	case 3:
		switch(select("Guardiao de Muralha (Templario):Colosso Reativo (Ferreiro)")) {
		case 1: jobchange Job_Crusader; break;
		case 2: jobchange Job_Blacksmith; break;
		}
		break;
	case 4:
		switch(select("Sacerdote Solar (Sacerdote):Arcanista Prismatico (Monge)")) {
		case 1: jobchange Job_Priest; break;
		case 2: jobchange Job_Monk; break;
		}
		break;
	case 5:
		switch(select("Ceifador do Vazio (Mago):Invocador do Abismo (Sabio)")) {
		case 1: jobchange Job_Wizard; break;
		case 2: jobchange Job_Sage; break;
		}
		break;
	}
	next;
	mes "[Mestre de Ascensao]";
	mes "Ascensao concluida. Que Lumen ou Umbra guiem voce.";
	close;
}

// 3. ARENA DE HONRA - PvP consensual por faixa de nivel
//    O jogador so entra se escolher e se estiver na faixa.
//    Voce deve criar os mapas (ex: pvp_n_1-1 ... pvp_n_1-5).
// ---------------------------------------------------------
prontera,162,185,5	script	Arena de Honra#eridan	4_M_BARD,{
	mes "[Arena de Honra]";
	mes "O PvP aqui e voluntario e entre iguais.";
	mes "Voce so luta com quem esta na sua faixa de nivel.";
	mes "Nenhum item e perdido ao ser derrotado.";
	next;
	if (BaseLevel < 15) {
		mes "[Arena de Honra]";
		mes "Novatos abaixo do nivel 15 nao participam.";
		close;
	}
	if (select("Entrar na arena:Cancelar") == 2) close;
	// Faixa de nivel: cada mapa recebe apenas jogadores da mesma faixa
	if (BaseLevel <= 30)      warp "pvp_n_1-1",0,0;
	else if (BaseLevel <= 50) warp "pvp_n_1-2",0,0;
	else if (BaseLevel <= 70) warp "pvp_n_1-3",0,0;
	else                      warp "pvp_n_1-4",0,0;
	end;
}

// 4. OPERACAO "CORACAO DE VIDRO" - Quest em grupo
//    Todos os membros do grupo online recebem a recompensa.
//    Troque os itens por itens custom do seu item_db.
// ---------------------------------------------------------
prontera,165,185,5	script	Cap. Rhea#eridan	4_M_ALCHE_A,{
	mes "[Capita Rhea]";
	mes "O Vigia-Zero tomou a torre de energia das Areias Vitreas.";
	mes "Traga um grupo de 4 a 6 Despertos.";
	next;
	if (getcharid(1) == 0) {
		mes "[Capita Rhea]";
		mes "Voce precisa estar em um grupo.";
		close;
	}
	if (select("Entregar a missao:Cancelar") == 2) close;
	if (getcharid(0) != getpartyleader(getcharid(1), 2)) {
		mes "[Capita Rhea]";
		mes "Apenas o lider do grupo entrega a missao.";
		close;
	}
	if (countitem("Crystal_Fragment") < 5) {
		mes "[Capita Rhea]";
		mes "Preciso de 5 Fragmentos de Cristal como prova.";
		close;
	}
	delitem "Crystal_Fragment", 5;
	.@leader = getcharid(3);
	getpartymember getcharid(1), 2;
	.@count = $@partymembercount;
	copyarray .@aid[0], $@partymemberaid[0], .@count;
	for (.@i = 0; .@i < .@count; .@i++) {
		if (isloggedin(.@aid[.@i])) {
			attachrid(.@aid[.@i]);
			getitem Old_Blue_Box, 1;
			getitem Yggdrasilberry, 3;
			set Zeny, Zeny + 50000;
			dispbottom "Operacao Coracao de Vidro concluida! Recompensa recebida.";
		}
	}
	attachrid(.@leader);
	mes "[Capita Rhea]";
	mes "A torre voltou a funcionar. Todos receberam a recompensa.";
	close;
}
